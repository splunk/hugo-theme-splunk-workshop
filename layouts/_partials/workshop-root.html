{{- /* workshop-root.html — given a page, return the section that scopes the workshop sidebar and the prev/next pager. The pager then DFS-walks this root's descendants, so prev/next flow across chapters within the workshop rather than dead-ending at chapter boundaries. Definitions (depth = path segments below the language home): - Language home (depth 0): /en/ - Depth-1 section: /en// - Pure hub: only sub-section children. No workshop root applies when this is the page being viewed; workshop root for descendants is the depth-2 ancestor. - Flat workshop: has direct .md page children. Workshop root = self. - Depth-2 section: /en/// — typical workshop root. - Depth ≥ 2: walk up to find the depth-2 ancestor (or depth-1 flat workshop) and return it. Returns the workshop root section, or `false` when the current page is a pure hub / the language home / the site home — the caller treats that as "render a hero layout, no sidebar". */ -}} {{- $page := . -}} {{- $root := false -}} {{- if and (not $page.IsHome) (ne $page.Kind "404") (ne $page.Kind "taxonomy") -}} {{- /* Candidate list = the page itself plus every ancestor up to (but not including) the language home. Hugo's .Ancestors gives us the parent chain ending at the language home; prepending $page lets us treat "the page IS the depth-1 section" as a candidate too. */ -}} {{- $candidates := slice $page | append $page.Ancestors -}} {{- /* Find the depth-1 candidate: the one whose parent IS the language home. There's only one match in the chain. */ -}} {{- $depthOne := false -}} {{- range $candidates -}} {{- if and .Parent .Parent.IsHome -}}{{- $depthOne = . -}}{{- end -}} {{- end -}} {{- if $depthOne -}} {{- /* Does $depthOne contain direct .md page children? Hugo 0.155+ made .Pages recursive, so we filter the returned regular pages to those whose .Parent is $depthOne itself. */ -}} {{- $depthOneHasDirectPages := false -}} {{- range where $depthOne.Pages "Kind" "page" -}} {{- if eq .Parent.RelPermalink $depthOne.RelPermalink -}} {{- $depthOneHasDirectPages = true -}} {{- end -}} {{- end -}} {{- if eq $depthOne $page -}} {{- /* Page is at depth 1 itself. Flat workshop iff it has direct page children; otherwise it's a hub (no sidebar). */ -}} {{- if $depthOneHasDirectPages -}}{{- $root = $page -}}{{- end -}} {{- else if $depthOneHasDirectPages -}} {{- /* Page is deeper, and $depthOne is a flat workshop — use it. */ -}} {{- $root = $depthOne -}} {{- else -}} {{- /* Page is deeper, $depthOne is a hub — workshop root is the depth-2 ancestor: the candidate whose parent is $depthOne. */ -}} {{- range $candidates -}} {{- if and .Parent (eq .Parent.RelPermalink $depthOne.RelPermalink) -}} {{- $root = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $root -}}
