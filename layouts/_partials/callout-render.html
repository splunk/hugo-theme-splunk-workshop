{{- /* Render a callout. Single source of truth for the callout markup, the icon SVG per type, the default title (resolved via i18n), and the optional collapsible
wrapper used by relearn-compatible notices. Usage: {{ partial "callout-render.html" (dict "type" "tip" // required: tip|note|info|warning|danger|success "title" "Custom title" // optional; falls back to i18n lookup of $type "inner" $renderedHTML // required: already piped through .Page.RenderString "icon" "rocket" // optional: Lucide icon name override "color" "#ff7700" // optional: custom accent colour (any CSS colour) "expanded" "true" // optional: "true"/"false"/"" — render as
"groupid" "advanced" // optional: synced-collapse group name "page" .Page) }} Output is intentionally flat (no internal line breaks inside an open tag) so Goldmark recognises the markup as an HTML block when the calling shortcode is invoked with the percent form (e.g. {{% notice %}}). */ -}} {{- $type := .type | default "note" -}} {{- $title := .title -}} {{- $inner := .inner -}} {{- $iconOv := .icon -}} {{- $color := .color -}} {{- $expR := .expanded -}} {{- $groupid := .groupid -}} {{- $icons := dict "tip" `` "note" `` "info" `` "warning" `` "danger" `` "success" `` -}} {{- $iconSVG := index $icons $type | default (index $icons "note") -}} {{- $stroke := cond (eq $type "success") "2.5" "2" -}} {{- /* Author-supplied callout titles support inline markdown (`**bold**`, `*italic*`, `` `code` ``, links). The `| string` cast handles callers that pass `template.HTML` (e.g. the render-blockquote hook forwards `.AlertTitle`, which is already-rendered) — markdownify refuses already-rendered values, but on a plain string it parses any markdown and lets raw HTML pass through. Avoid leading digit-dot-space in titles — Goldmark will treat it as an ordered-list start and emit
  1. ...
. */ -}} {{- $resolvedTitle := $title | default (i18n $type) | default (humanize $type) | string | markdownify -}} {{- /* expanded mode: render as
only when explicitly set. */ -}} {{- $useDetails := false -}} {{- $detailsOpen := false -}} {{- if $expR -}} {{- $e := lower (trim (string $expR) " ") -}} {{- if ne $e "" -}} {{- $useDetails = true -}} {{- $detailsOpen = eq $e "true" -}} {{- end -}} {{- end -}} {{- $cls := printf "callout callout--%s" $type -}} {{- if $useDetails }}{{ $cls = printf "%s callout--expandable" $cls }}{{ end -}} {{- $iconHTML := "" -}} {{- if $iconOv -}} {{- $iconHTML = partial "icon-svg.html" $iconOv -}} {{- else -}} {{- $iconHTML = printf `` $stroke (string $iconSVG) -}} {{- end -}} {{- $styleAttr := "" -}} {{- with $color -}} {{- $safeColor := partial "css-color.html" (dict "value" . "fallback" "") -}} {{- if $safeColor }}{{ $styleAttr = printf ` style="--callout-color: %s"` ($safeColor | safeCSS) }}{{ end -}} {{- end -}} {{- if $useDetails -}}
{{ $iconHTML | safeHTML }}{{ $resolvedTitle }}
{{ $inner | safeHTML }}
{{- else -}}
{{ $iconHTML | safeHTML }}

{{ $resolvedTitle }}

{{ $inner | safeHTML }}
{{- end -}}
