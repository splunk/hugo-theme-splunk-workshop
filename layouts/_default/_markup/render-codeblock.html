{{- /* Code-block render hook. Supports a `file` attribute on the fence, e.g.: ```js {file="hello.js"} ```mermaid fences render as diagrams (mirrors the mermaid shortcode). */ -}} {{- $lang := or .Type "text" -}} {{- if eq $lang "mermaid" -}} {{- .Page.Store.Set "_needs_mermaid" true -}} {{- $inner := trim .Inner "\n\r" | replaceRE `\n\s*\n` "\n" -}} 
{{- $inner | htmlEscape | safeHTML -}}
{{- else -}} {{- $file := .Attributes.file -}} {{- $hl := .Attributes.hl_lines -}} {{- $opts := dict "lineNos" false "tabWidth" 2 -}} {{- if $hl }}{{ $opts = merge $opts (dict "hl_lines" $hl) }}{{ end -}}
{{ if $file }}{{ $file }}{{ else }}{{ $lang }}{{ end }}
{{ highlight (trim .Inner "\n\r") $lang $opts }}
{{- end -}}
