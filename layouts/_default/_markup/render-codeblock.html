{{- /* Code-block render hook. Supports a `file` attribute on the fence, e.g.: ```js {file="hello.js"} Renders a header with filename, language, and copy button. */ -}} {{- $lang := or .Type "text" -}} {{- $file := .Attributes.file -}} {{- $hl := .Attributes.hl_lines -}} {{- $opts := dict "lineNos" false "tabWidth" 2 -}} {{- if $hl }}{{ $opts = merge $opts (dict "hl_lines" $hl) }}{{ end -}}
{{ if $file }}{{ $file }}{{ else }}{{ $lang }}{{ end }}
{{ highlight (trim .Inner "\n\r") $lang $opts }}
