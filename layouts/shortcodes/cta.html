{{- /* CTA — the big Splunk-themed call-to-action pill (gradient fill, generous padding, shadow). Used for primary actions on landing pages, hero blocks, and end-of-workshop next-steps. For inline page-content buttons reach for {{< button >}} instead — that's the smaller relearn-compatible variant. Usage: {{< cta href="/install/" icon="download" >}}Install now{{< /cta >}} {{< cta href="https://example.com" target="_blank" style="ghost" >}}Open{{< /cta >}} $style accepts: primary (default — gradient pill, big shadow) secondary | ghost | transparent (outlined variant) */ -}} {{- $hrefRaw := .Get "href" -}} {{- $href := partial "site-href.html" $hrefRaw -}} {{- $style := .Get "style" | default "primary" -}} {{- $styleKey := lower (string $style) -}} {{- $icon := .Get "icon" -}} {{- $iconpos := .Get "iconposition" | default "left" -}} {{- $target := .Get "target" -}} {{- $title := .Get "title" -}} {{- $cls := "btn btn--primary" -}} {{- if or (eq $styleKey "secondary") (eq $styleKey "ghost") (eq $styleKey "transparent") -}} {{- $cls = "btn btn--ghost" -}} {{- end -}} {{- $body := .Inner | .Page.RenderString -}} {{- if not (trim ($body | string) " \n\r\t") }}{{ $body = $title }}{{ end -}} {{- $tag := "a" -}} {{- if not $href }}{{ $tag = "button" }}{{ end -}} {{- $iconHTML := "" -}} {{- if $icon -}} {{- $iconHTML = printf `%s` (partial "icon-svg.html" $icon) -}} {{- end -}} {{- /* Single-line output so the CTA can sit inline inside a paragraph without markdown splitting at internal line-breaks. */ -}} {{- if eq $tag "a" -}} {{ if and $icon (eq $iconpos "left") }}{{ $iconHTML | safeHTML }}{{ end }}{{ $body | safeHTML }}{{ if and $icon (eq $iconpos "right") }}{{ $iconHTML | safeHTML }}{{ end }}{{ if not $icon }}→{{ end }} {{- else -}} {{- end -}}
