{{- /* Tab — child of tabs. When used standalone (no parent {{< tabs >}}) it renders as a single labeled block, matching relearn's behaviour. Args: title | name | label (any of the three is accepted as the tab label; positional 0 also works). `name` is the relearn idiom and is kept here for migrated content. `icon` is optional. */ -}} {{- $label := .Get "title" | default (.Get "name") | default (.Get "label") | default (.Get 0) | default "Tab" -}} {{- $icon := .Get "icon" -}} {{- $content := .Inner | .Page.RenderString -}} {{- if and .Parent (eq .Parent.Name "tabs") -}} {{- /* Nested inside tabs — push into parent's bucket. Key on the parent tabs shortcode's .Position so the bucket is unique per tabs instance even when the parent tabs is re-rendered inside another percent shortcode's .Inner (see tabs.html for the same fix). */ -}} {{- $posKey := printf "%v" .Parent.Position -}} {{- $idSlug := replaceRE "[^a-zA-Z0-9]" "-" $posKey -}} {{- $key := printf "_tabs_%s" $idSlug -}} {{- $entry := dict "label" $label "icon" $icon "content" $content -}} {{- .Page.Store.Add $key (slice $entry) -}} {{- else -}} {{- /* Standalone — render as a labeled single-pane block. Flat output so indented HTML isn't parsed as a code block when nested in a percent shortcode (see tabs.html for the same constraint). */ -}}
{{ $content | safeHTML }}
{{- end -}}
