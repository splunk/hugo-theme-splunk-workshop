{{- /* YouTube embed — accepts the video ID as the first arg. {{< youtube "dQw4w9WgXcQ" >}} {{< youtube id="abc123" title="Intro" >}} */ -}} {{- $id := .Get "id" | default (.Get 0) -}} {{- $title := .Get "title" | default "YouTube video" -}} {{- if not $id -}} {{- errorf "{{< youtube >}} called without a video id (page %q)" .Page.Path -}} {{- else if not (findRE `^[A-Za-z0-9_-]{11}$` $id) -}} {{- errorf "{{< youtube >}}: invalid video id %q in page %q — expected 11 chars of [A-Za-z0-9_-]" $id .Page.Path -}} {{- end -}}
