{{- /* Relearn-compatible include — embed another page's content. Args: file - path under content/, e.g. "shared/footer.md" hidefirstheading - "true" to drop the first H1 */ -}} {{- $file := .Get "file" | default (.Get 0) -}} {{- $hide := eq (.Get "hidefirstheading" | default (.Get 1) | default "false") "true" -}} {{- with $.Page.GetPage $file -}} {{- $html := .RenderShortcodes -}} {{- $rendered := $html | $.Page.RenderString -}} {{- if $hide -}} {{- /* Split at the first and discard everything up to and including it. We deliberately operate on RENDERED HTML (not .RawContent) so that Goldmark has already escaped any `` inside code fences to `</h1>` — that makes the split safer than a markdown- level regex, which would need its own code-fence awareness to avoid false matches like `# shell comment` inside ```bash blocks. Goldmark always emits lowercase tags, so `` never appears. */ -}} {{- $parts := split ($rendered | string) "" -}} {{- if gt (len $parts) 1 -}} {{- $rendered = strings.Join (after 1 $parts) "" -}} {{- end -}} {{- end -}} {{ $rendered | safeHTML }} {{- else -}} {{- errorf "{{< include >}} could not find page %q" $file -}} {{- end -}}
