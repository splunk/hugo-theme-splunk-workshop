{{- /* Relearn-compatible include — embed another page's content. Args: file - path under content/, e.g. "shared/footer.md" hidefirstheading - "true" to drop the first H1 */ -}} {{- $file := .Get "file" | default (.Get 0) -}} {{- $hide := eq (.Get "hidefirstheading" | default (.Get 1) | default "false") "true" -}} {{- with $.Page.GetPage $file -}} {{- $html := .RenderShortcodes -}} {{- $rendered := $html | $.Page.RenderString -}} {{- if $hide -}} {{- /* Split at the first and discard everything up to and including it. Robust to nested HTML inside the heading (Goldmark won't generate a literal "" inside an h1, and code blocks render entity-escaped — `</h1>` — so a fake `` is implausible). */ -}} {{- $parts := split ($rendered | string) "" -}} {{- if gt (len $parts) 1 -}} {{- $rendered = strings.Join (after 1 $parts) "" -}} {{- end -}} {{- end -}} {{ $rendered | safeHTML }} {{- else -}} {{- errorf "{{< include >}} could not find page %q" $file -}} {{- end -}}
