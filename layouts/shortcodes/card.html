{{- /* Card — renders one card. Always inline; if used inside `{{< cards >}}` the parent's CSS grid lays it out, otherwise it stands alone. Canonical API: positional 0 = title positional 1 = href named: title, href, icon, image, hero-icon, color, show-time, show-pages `icon` is the small inline glyph next to the title; `hero-icon` renders a large Lucide icon on the gradient header band. When `hero-icon` is set (or a raster image is shown), the inline `icon` is suppressed — one identity signal per card. `show-time` / `show-pages` (booleans, default false) opt into a meta row pulled from the linked Hugo page (reading time, child-page count). Omit either flag or set it to false to suppress that value. Both no-op for external `href` values where the target can't be resolved. Markup is emitted by `cards/shortcode-card.html` — the same partial `{{< children type="card" >}}` uses. Legacy alias (deprecated, accepted for relearn compatibility): `url` — use `href` instead. */ -}} {{- $title := .Get "title" | default (.Get 0) -}} {{- $hrefRaw := .Get "href" | default (.Get "url") | default (.Get 1) -}} {{- $href := partial "site-href.html" $hrefRaw -}} {{- $icon := .Get "icon" -}} {{- $image := .Get "image" -}} {{- $heroIcon := .Get "hero-icon" -}} {{- $color := .Get "color" -}} {{- if $color }}{{- $color = partial "css-color.html" (dict "value" $color "fallback" "var(--color-accent)") -}}{{- end -}} {{- $showTime := eq (lower (.Get "show-time" | default "" | string)) "true" -}} {{- $showPages := eq (lower (.Get "show-pages" | default "" | string)) "true" -}} {{- $content := .Inner | .Page.RenderString -}} {{- /* Resolve the link target once — used for both the image auto-pull fallback and the optional meta row. Nil for external hrefs. */ -}} {{- $tp := "" -}} {{- with $hrefRaw }}{{ $tp = site.GetPage . }}{{ end -}} {{- /* Final URL for the card image. Resolution order: 1. Explicit `image="..."` arg — author-provided, wins. 2. Auto-pull: target page's `images[0]` via page-image.html. 3. Nothing — render without an image (or fall through to hero-icon). */ -}} {{- $imgURL := "" -}} {{- if $image -}} {{- $imgURL = $image | relURL -}} {{- else if $tp -}} {{- $imgURL = partial "cards/page-image.html" $tp -}} {{- end -}} {{- $count := 0 -}} {{- if $tp -}}{{- $count = partial "workshop/children-count.html" $tp -}}{{- end -}} {{- $meta := slice -}} {{- if and $showTime $tp -}} {{- with $tp.Params.time | default $tp.Params.duration | default (printf "%d min" $tp.ReadingTime) -}} {{- $meta = $meta | append . -}} {{- end -}} {{- end -}} {{- if and $showPages $tp -}} {{- if gt $count 1 -}}{{ $meta = $meta | append (i18n "pagesCount" (dict "Count" $count)) }}{{- end -}} {{- end -}} {{- $trackRoot := "" -}} {{- with $tp }}{{ $trackRoot = .RelPermalink }}{{ end -}} {{- /* Flag the page so layouts (notably index.html) suppress their auto-card-grid when the body has already authored its own. */ -}} {{- .Page.Store.Set "_has_cards" true -}} {{- partial "cards/shortcode-card.html" (dict "title" $title "href" $href "color" $color "imgURL" $imgURL "heroIcon" $heroIcon "inlineIcon" $icon "body" $content "meta" $meta "trackRoot" $trackRoot "trackLeaf" (and (ne $trackRoot "") (eq $count 0)) ) -}}
