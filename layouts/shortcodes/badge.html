{{- /* Badge — relearn-compatible. Args: color - hex / CSS color, OR a preset name (accent | info | success | warn | danger) style - alias of color title - text shown to the left of content icon - optional icon content - .Inner OR positional 0 Backward-compatible: positional 0=text, 1=type */ -}} {{- $text := .Inner | string -}} {{- if not $text }}{{ $text = .Get "content" | default (.Get "text") | default (.Get 0) | default "" }}{{ end -}} {{- $color := .Get "color" | default (.Get "style") | default (.Get 1) -}} {{- $title := .Get "title" -}} {{- $icon := .Get "icon" -}} {{- $known := slice "accent" "info" "success" "warn" "warning" "danger" "primary" "secondary" -}} {{- $cls := "badge" -}} {{- $inline := "" -}} {{- if $color -}} {{- if in $known $color -}} {{- $alias := $color -}} {{- if eq $alias "warning" }}{{ $alias = "warn" }}{{ end -}} {{- if eq $alias "primary" }}{{ $alias = "accent" }}{{ end -}} {{- if eq $alias "secondary" }}{{ $alias = "" }}{{ end -}} {{- if $alias }}{{ $cls = printf "badge badge--%s" $alias }}{{ end -}} {{- else -}} {{- /* Custom colour: allow-list the value before interpolating it into a multi-property inline style. `safeCSS` marks the string trusted but does NOT validate it, so an unguarded value like `red;background:url(...)` would inject extra declarations. Same defence as `textcolor.html` — restrict to a conservative char-set (hex, names, rgb/hsl/var/calc) and reject `;` `:` `{` `}` quotes. A failing value falls back to currentColor so the badge degrades safely rather than rendering attacker-controlled CSS. */ -}} {{- if not (findRE `^[a-zA-Z0-9#.,()% /_-]+$` $color) }}{{ $color = "currentColor" }}{{ end -}} {{- $inline = printf "color:%s;border-color:color-mix(in oklab, %s 40%%, transparent);background:color-mix(in oklab, %s 10%%, transparent)" $color $color $color -}} {{- end -}} {{- end -}} {{- with $icon }}{{ partial "icon-svg.html" . }}{{ end -}} {{- with $title }}{{ . }}{{ end -}} {{ $text | safeHTML -}}
