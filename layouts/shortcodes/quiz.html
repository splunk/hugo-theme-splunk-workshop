{{- /* Quiz — multiple choice with reveal-on-click feedback. Usage: {{< quiz question="Which command counts events?" >}} {{< quiz-option correct=true >}}stats count{{< /quiz-option >}} {{< quiz-option >}}timechart{{< /quiz-option >}} {{< quiz-feedback >}}`stats count` aggregates events.{{< /quiz-feedback >}} {{< /quiz >}} The markup is one line on purpose. A quiz often sits inside {{% notice %}} or {{% exercise %}}, and those percent-form shortcodes run their inner HTML through Markdown again. An indented option, or a whitespace-only line before it, makes Goldmark turn the buttons into a code block. */ -}} {{- $q := .Get "question" | default (.Get 0) | default (i18n "quizQuestion") -}} {{- $optsKey := printf "_quiz_opts_%d" .Ordinal -}} {{- $fbKey := printf "_quiz_fb_%d" .Ordinal -}} {{- $_ := .Inner -}} {{- $opts := .Page.Store.Get $optsKey | default slice -}} {{- $fb := .Page.Store.Get $fbKey -}} {{- $items := slice -}} {{- range $i, $o := $opts -}} {{- $text := $o.text | replaceRE "\n+" "" -}} {{- $items = $items | append (printf "
    • " $o.correct (printf "%c" (add 65 $i)) $text) -}} {{- end -}} {{- $feedback := "" -}} {{- with $fb -}} {{- $feedback = printf "
    %s
    " (replaceRE "\n+" "" .) -}} {{- end -}}

      {{ delimit $items "" | safeHTML }}
