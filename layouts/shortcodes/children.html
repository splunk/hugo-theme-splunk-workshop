{{- /* Relearn-compatible children — list child pages of the current section. Args: page - section path (default: current page) type - "card" | "list" | "h2" | "h3" | "h4" (relearn canonical) style - alias of type depth - how deep to recurse (default 1) — currently shallow only description - "true" to include description (default true) image - "true" to render each child's featured image (params.images[0]) as a banner on top of card-style children (relearn compat). Image wins over the child's `icon` hero; without an image the `icon` front-matter key renders as a hero icon. showhidden - "true" to include hidden pages (default false) sort - weight | title (default weight) Card markup is emitted by `cards/shortcode-card.html` — the same partial `{{< card >}}` uses — so both paths share typography, media, and meta. IMPORTANT: this template emits HTML on a single contiguous block with no leading whitespace, so it works inside `{{% children %}}` (percent form, which re-runs Goldmark and would otherwise treat indented HTML as a code block). */ -}} {{- $page := .Get "page" -}} {{- $type := .Get "type" | default (.Get "style") | default "list" -}} {{- $depth := int (.Get "depth" | default 1) -}} {{- $desc := ne (.Get "description" | default "true") "false" -}} {{- $img := eq (.Get "image" | default "false") "true" -}} {{- $sort := .Get "sort" | default "weight" -}} {{- $hide := ne (.Get "showhidden" | default "false") "true" -}} {{- /* notime="true" suppresses the time pill on every card in this listing. Useful when the listing groups reference content or short pages where a duration estimate adds noise. Other meta items (difficulty, page count) keep rendering. */ -}} {{- $notime := eq (.Get "notime" | default "false") "true" -}} {{- $start := .Page -}} {{- if $page -}} {{- with $.Page.GetPage $page }}{{ $start = . }}{{ end -}} {{- end -}} {{- $items := $start.Pages -}} {{- if $start.Sections }}{{ $items = $items | union $start.Sections }}{{ end -}} {{- if $hide }}{{ $items = where $items "Params.hidden" "!=" true }}{{ end -}} {{- if eq $sort "title" }}{{ $items = $items.ByTitle }}{{ else }}{{ $items = $items.ByWeight }}{{ end -}} {{- if eq $type "card" -}} {{- /* Flag the page so layouts (notably index.html) suppress their auto-card-grid when the body has already authored cards. */ -}} {{- .Page.Store.Set "_has_cards" true -}}
{{- range $items -}} {{- $child := . -}} {{- $imgURL := "" -}} {{- if $img }}{{ $imgURL = partial "cards/page-image.html" $child }}{{ end -}} {{- $heroIcon := "" -}} {{- if not $imgURL }}{{ with $child.Params.icon }}{{ $heroIcon = . }}{{ end }}{{ end -}} {{- $body := "" -}} {{- if and $desc $child.Description }}{{ $body = printf "
