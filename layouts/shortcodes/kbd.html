{{- /* Keyboard key — accepts one or more keys joined by + or ,. Usage: {{< kbd "Cmd+Shift+P" >}} or {{< kbd "Ctrl" "K" >}} */ -}} {{- $raw := .Get 0 -}} {{- if not $raw -}} {{- $parts := slice -}} {{- range seq 0 5 -}}{{- with $.Get . }}{{- $parts = $parts | append . -}}{{- end -}}{{- end -}} {{- $raw = delimit $parts "+" -}} {{- end -}} {{- $keys := split (replace $raw "," "+") "+" -}} {{- range $i, $k := $keys -}} {{- if $i }}+{{ end -}} {{ trim $k " " }} {{- end -}}
