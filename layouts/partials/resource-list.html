{{- /* Shared body for the `attachments` and `resources` shortcodes. Renders the page's non-image resources as a downloadable list. Both shortcodes call this with their own default title. Args (dict): page - the Hugo Page whose .Resources we list. title - heading text. pattern - optional regex filter on resource Name. sort - "asc" or "desc" (default asc). icon - icon-svg name to prefix each row (default "file"). */ -}} {{- $page := .page -}} {{- $title := .title -}} {{- $pattern := .pattern -}} {{- $sort := .sort | default "asc" -}} {{- $icon := .icon | default "file" -}} {{- $items := $page.Resources -}} {{- if $pattern -}} {{- /* Hugo's `where` has no regexp operator — filter manually. */ -}} {{- $filtered := slice -}} {{- range $items -}} {{- if findRE $pattern .Name -}}{{- $filtered = $filtered | append . -}}{{- end -}} {{- end -}} {{- $items = $filtered -}} {{- end -}} {{- $items = where $items "ResourceType" "ne" "image" -}} {{- $items = sort $items "Title" -}} {{- if eq $sort "desc" }}{{ $items = sort $items "Title" "desc" }}{{ end -}} {{- if $items }}

{{ $title }}

    {{ range $items }}
  • {{ partial "icon-svg.html" $icon }} {{ .Title | default .Name }}
    • {{ end }}
{{ end -}}
