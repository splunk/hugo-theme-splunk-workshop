{{- /* Resolve an href against the site baseURL. Hugo's relURL preserves a leading "/" (treats it as domain-rooted), which breaks project pages where baseURL has a path prefix (e.g. GitHub Pages at https://owner.github.io/repo/). This helper strips the leading slash before piping through relURL so internal paths get the prefix, while passing external (http*) and anchor (#) hrefs through unchanged. Usage: href="{{ partial "site-href.html" $someHref }}" */ -}} {{- $raw := . -}} {{- $out := $raw -}} {{- if $raw -}} {{- if not (or (hasPrefix $raw "http") (hasPrefix $raw "#")) -}} {{- $out = (strings.TrimPrefix "/" $raw) | relURL -}} {{- end -}} {{- end -}} {{- return $out -}}
