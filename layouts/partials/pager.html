{{- /* Workshop prev/next pager. Default behaviour: builds a flat DFS list of every page within the page's FirstSection (or the highest ancestor with `layout = "chapter"`), finds the current position, and links to its neighbours. Override: the `pager` shortcode can write {url,title} dicts to Scratch for either side, and we use those instead. The auto-DFS only runs for sides that weren't overridden, so authors can override Next while keeping Prev auto, or both, or neither. Skipped on the home page and when `nopager: true` is in front matter. */ -}} {{- if .IsHome -}}{{- return -}}{{- end -}} {{- if eq (.Params.nopager | default false) true -}}{{- return -}}{{- end -}} {{- $prev := false -}}{{- $next := false -}} {{- /* Shortcode overrides — already {url, title} dicts. */ -}} {{- with .Page.Scratch.Get "pager_override_prev" -}}{{- $prev = . -}}{{- end -}} {{- with .Page.Scratch.Get "pager_override_next" -}}{{- $next = . -}}{{- end -}} {{- /* Auto-DFS for any side that wasn't overridden. */ -}} {{- if or (not $prev) (not $next) -}} {{- $root := partial "workshop-root.html" . -}} {{- $flat := partial "flat-pages.html" $root -}} {{- /* Always include the scope root itself as index 0: - On the chapter landing, idx=0 → no Previous, Next = first child. - On the first child page, idx=1 → Previous = chapter landing. - On the last child page, no Next (end of scope, no wrap). */ -}} {{- $flat = (slice $root) | append $flat -}} {{- $idx := -1 -}} {{- range $i, $p := $flat -}} {{- if and (eq $idx -1) (eq $p.RelPermalink $.RelPermalink) -}}{{- $idx = $i -}}{{- end -}} {{- end -}} {{- if and (not $prev) (gt $idx 0) -}} {{- $p := index $flat (sub $idx 1) -}} {{- $prev = dict "url" $p.RelPermalink "title" $p.LinkTitle -}} {{- end -}} {{- if and (not $next) (ge $idx 0) (lt $idx (sub (len $flat) 1)) -}} {{- $p := index $flat (add $idx 1) -}} {{- $next = dict "url" $p.RelPermalink "title" $p.LinkTitle -}} {{- end -}} {{- end -}} {{- if or $prev $next -}} {{- /* Class hooks the CSS to collapse the grid down to one column and align the single remaining button to the correct edge — no empty placeholder cells. */ -}} {{- $cls := "pager" -}} {{- if and $prev (not $next) }}{{ $cls = "pager pager--end" }}{{ end -}} {{- if and $next (not $prev) }}{{ $cls = "pager pager--start" }}{{ end -}}
{{ with $prev }} {{ i18n "previous" }} {{ .title }} {{ end }} {{ with $next }} {{ i18n "next" }} {{ .title }} {{ end }}
{{- end -}}
