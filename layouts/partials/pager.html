{{- /* Workshop prev/next pager. Builds a flat DFS list of every page within the page's FirstSection (or the highest ancestor with `layout = "chapter"` if one is set), finds the current position, and links to its neighbours. Skipped on the home page. */ -}} {{- if .IsHome -}}{{- return -}}{{- end -}} {{- if eq (.Params.nopager | default false) true -}}{{- return -}}{{- end -}} {{- $root := partial "workshop-root.html" . -}} {{- $flat := partial "flat-pages.html" $root -}} {{- /* Always include the scope root itself as index 0 of the flat list: - On the chapter landing, idx=0 → no Previous, Next = first child. - On the first child page, idx=1 → Previous = chapter landing. - On the last child page, no Next (end of scope, no cross-chapter wrap). */ -}} {{- $flat = (slice $root) | append $flat -}} {{- $idx := -1 -}} {{- range $i, $p := $flat -}} {{- if and (eq $idx -1) (eq $p.RelPermalink $.RelPermalink) -}}{{- $idx = $i -}}{{- end -}} {{- end -}} {{- $prev := false -}}{{- $next := false -}} {{- if gt $idx 0 -}}{{- $prev = index $flat (sub $idx 1) -}}{{- end -}} {{- if and (ge $idx 0) (lt $idx (sub (len $flat) 1)) -}}{{- $next = index $flat (add $idx 1) -}}{{- end -}} {{- if or $prev $next -}} {{- /* Class hooks the CSS to collapse the grid down to one column and align the single remaining button to the correct edge — no empty placeholder cells. */ -}} {{- $cls := "pager" -}} {{- if and $prev (not $next) }}{{ $cls = "pager pager--end" }}{{ end -}} {{- if and $next (not $prev) }}{{ $cls = "pager pager--start" }}{{ end -}}
{{ with $prev }} {{ i18n "previous" }} {{ .LinkTitle }} {{ end }} {{ with $next }} {{ i18n "next" }} {{ .LinkTitle }} {{ end }}
{{- end -}}
