Each item links to the equivalent translation of the current page when one exists (.Translations); otherwise it falls back to the language's home page. The current language is rendered as a non-link button with aria-current. */ -}} {{- if gt (len site.Languages) 1 -}} {{- $page := . -}}
    {{- range site.Languages -}} {{- $code := .Lang -}} {{- $name := .Label | default .Lang -}} {{- $isActive := eq $code $page.Site.Language.Lang -}} {{- /* Pick the matching translation if it exists, else fall back to the language's home page so the user lands somewhere sensible. */ -}} {{- $href := "" -}} {{- range $page.AllTranslations -}} {{- if eq .Lang $code }}{{ $href = .RelPermalink }}{{ end -}} {{- end -}} {{- if not $href -}} {{- $href = printf "/%s/" $code | relURL -}} {{- end -}}
  • {{ $name }}

{{- end -}}
